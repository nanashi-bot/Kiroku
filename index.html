// 月と日を作る const monthSelect = document.getElementById("birthMonth"); const daySelect = document.getElementById("birthDay"); for (let i = 1; i <= 12; i++) { const option = document.createElement("option"); option.value = i; option.textContent = i + "月"; monthSelect.appendChild(option); } for (let i = 1; i <= 31; i++) { const option = document.createElement("option"); option.value = i; option.textContent = i + "日"; daySelect.appendChild(option); }
